Market Baseline

Introduction

For this market baseline, we are using job-posting data from Computer Systems Design and Related Services industry (NAICS 5415). We look at job titles, locations, work arrangements, salaries, and analytics tools mentioned in the job descriptions. This will help us understand the opportunities shown in our data and give us a starting point for comparing them with our team’s skills.

We began by checking how much data we have and what information is available. We then looked at missing values, converted usable salary ranges into annual amounts, and created charts to make the patterns easier to understand. We found some limitations along the way, including missing location information, unknown work arrangements, and roles outside our intended analytics focus.

from pyspark.sql import functions as F

has_annual = (
    F.col("annual_salary_min").isNotNull()
    & F.col("annual_salary_max").isNotNull()
)

has_hourly = (
    F.col("hourly_salary_min").isNotNull()
    & F.col("hourly_salary_max").isNotNull()
)

currency_code = F.regexp_extract(
    F.upper(F.coalesce(F.col("salary_text"), F.lit(""))),
    r"\b(USD|CAD|EUR|GBP|AUD|NZD|INR|SGD|JPY|CHF|CNY|AED)\b",
    1
)

jobs_pay = (
    jobs_clean
    .withColumn(
        "annualized_min",
        F.when(has_annual, F.col("annual_salary_min"))
         .when(has_hourly, F.col("hourly_salary_min") * 2080)
    )
    .withColumn(
        "annualized_max",
        F.when(has_annual, F.col("annual_salary_max"))
         .when(has_hourly, F.col("hourly_salary_max") * 2080)
    )
    .withColumn(
        "salary_basis",
        F.when(has_annual, "Reported annual")
         .when(has_hourly, "Hourly converted × 2080")
         .otherwise("No complete range")
    )
    .withColumn(
        "salary_currency",
        F.when(currency_code != "", currency_code)
         .otherwise("Unconfirmed")
    )
)

jobs_pay.filter(F.col("annualized_min").isNotNull()) \
    .groupBy("salary_currency", "salary_basis").count() \
    .orderBy(F.desc("count")) \
    .show(truncate=False)

# Identify US locations from country labels or state abbreviations.
us_states = (
    "AL|AK|AZ|AR|CA|CO|CT|DE|FL|GA|HI|ID|IL|IN|IA|KS|KY|LA|"
    "ME|MD|MA|MI|MN|MS|MO|MT|NE|NV|NH|NJ|NM|NY|NC|ND|OH|OK|"
    "OR|PA|RI|SC|SD|TN|TX|UT|VT|VA|WA|WV|WI|WY|DC"
)

location = F.upper(F.coalesce(F.col("location_text"), F.lit("")))

us_location = (
    location.rlike(r"\bUNITED STATES\b|\bUSA\b|\bU\.S\.A?\.")
    | location.rlike(r",\s*(" + us_states + r")\b")
    | location.rlike(
        r"\b(ARIZONA|TEXAS|CALIFORNIA|FLORIDA|MARYLAND|"
        r"MICHIGAN|MISSOURI|NEW YORK)\b"
    )
    | location.rlike(r"\bLOS ANGELES METROPOLITAN AREA\b")
)

assume_usd = (
    (F.col("salary_currency") == "Unconfirmed")
    & us_location
)

jobs_pay = (
    jobs_pay
    .withColumn(
        "currency_basis",
        F.when(assume_usd, "Assumed from US location")
        .when(
            F.col("salary_currency") != "Unconfirmed",
            "Explicit currency in salary text"
        )
        .otherwise("Unconfirmed")
    )
    .withColumn(
        "salary_currency",
        F.when(assume_usd, "USD")
        .otherwise(F.col("salary_currency"))
    )
)

jobs_pay.filter(
    F.col("annualized_min").isNotNull()
).groupBy(

tool_patterns = [
    ("SQL", r"\bsql\b"),
    ("Python", r"\bpython\b"),
  ("Excel", r"\bexcel\b(?!\s+(?:in|at|as|within)\b)"),
    ("Tableau", r"\btableau\b"),
    ("Power BI", r"\bpower[\s-]*bi\b"),
    ("Apache Spark / PySpark", r"\b(?:apache\s+spark|pyspark)\b"),
    ("Pandas", r"\bpandas\b")
]

description_jobs = jobs_clean.filter(has_description).withColumn(
    "description_lower",
    F.lower(F.col("description"))
)

description_count = description_jobs.select("job_id").distinct().count()

tool_mentions = (
    description_jobs
    .select(
        "job_id",
        F.explode(F.array(*[
            F.struct(
                F.lit(tool).alias("tool"),
                F.col("description_lower").rlike(pattern).alias("mentioned")
            )
            for tool, pattern in tool_patterns
        ])).alias("match")
    )
    .groupBy(F.col("match.tool").alias("tool"))
    .agg(
        F.countDistinct(
            F.when(F.col("match.mentioned"), F.col("job_id"))
        ).alias("postings_mentioning_tool")
    )
    .withColumn(
        "percent_of_descriptions",
        F.round(
            F.col("postings_mentioning_tool") / description_count * 100, 1
        )
    )
    .orderBy(F.desc("postings_mentioning_tool"), "tool")
)

tool_mentions.show(truncate=False)

For the 441 postings included in this salary analysis, the median annualized salary midpoint was $134,250. The distribution has a tail toward higher salaries, showing that some advertised ranges were much higher than others. These are advertised range midpoints rather than actual earnings, and the sample combines different roles and experience levels. Hourly salaries were converted using 2,080 hours per year, and some currencies were assumed to be USD.
