jobs_clean = spark.sql("""

    WITH

    -- ── STEP 1: Remove noisy data ──────────────────────────────────────────
    -- Rows with null job_id are dropped; they cannot be keyed or joined.
    -- Rows with null title are dropped; they are unusable for downstream analysis.
    
    base AS (
        SELECT *
        FROM job_facts_raw
        WHERE job_id IS NOT NULL
          AND title IS NOT NULL
    ),

    -- ── STEP 2: Normalize categorical labels ─────────────────────────
    -- remote_status and employment_type arrive with inconsistent casing
    -- and freeform strings (e.g. "REMOTE", "Remote Work", "wfh").
    -- Map to four canonical values. 
    -- Any unrecognized value defaults to 'Unknown' rather than NULL so aggregations are more meaningful.
    labeled AS (
        SELECT *,

        CASE
            WHEN LOWER(TRIM(remote_status)) IN ('remote', 'fully remote', 'work from home', 'wfh', 'remote work')
                OR LOWER(TRIM(remote_status)) LIKE '%remote%' AND LOWER(TRIM(remote_status)) NOT LIKE '%partial%'
                OR LOWER(TRIM(remote_status)) LIKE '%wfh%'
                OR LOWER(TRIM(remote_status)) LIKE '%home%'
                THEN 'Remote'
            WHEN LOWER(TRIM(remote_status)) IN ('hybrid', 'hybrid remote', 'partial remote', 'hybrid/remote')
                OR LOWER(TRIM(remote_status)) LIKE '%hybrid%'
                OR LOWER(TRIM(remote_status)) LIKE '%partial%'
                THEN 'Hybrid'
            WHEN LOWER(TRIM(remote_status)) IN ('on-site', 'onsite', 'in-office', 'in office', 'on site', 'office')
                OR LOWER(TRIM(remote_status)) LIKE '%office%'
                OR LOWER(TRIM(remote_status)) LIKE '%site%'
                THEN 'On-site'
            ELSE 'Unknown'
        END AS remote_status_clean,

        CASE
            WHEN LOWER(TRIM(employment_type)) IN ('full-time', 'full time', 'fulltime', 'ft', 'permanent')
                OR LOWER(TRIM(employment_type)) LIKE '%permanent%'
                OR LOWER(TRIM(employment_type)) LIKE '%full%'
                THEN 'Full-time'
            WHEN LOWER(TRIM(employment_type)) IN ('part-time', 'part time', 'parttime', 'pt')
                OR LOWER(TRIM(employment_type)) LIKE '%part%'
                THEN 'Part-time'
            WHEN LOWER(TRIM(employment_type)) IN ('contract', 'contractor', 'temp', 'temporary', 'freelance', 'c2c')
                OR LOWER(TRIM(employment_type)) LIKE '%contract%'
                OR LOWER(TRIM(employment_type)) LIKE '%temp%'
                OR LOWER(TRIM(employment_type)) LIKE '%freelance%'
                THEN 'Contract'
            WHEN LOWER(TRIM(employment_type)) IN ('internship', 'intern', 'co-op', 'coop')
                OR LOWER(TRIM(employment_type)) LIKE '%intern%'
                THEN 'Internship'
            ELSE 'Unknown'
        END AS employment_type_clean


        FROM base
    ),

    -- ── STEP 3: Salary outlier handling ─────────────────────────────
    -- Salary fields contain a few data errors: $0, $1, implausibly high values from mis-parsed hourly-to-annual conversions, or potentially test records. 
    -- Values outside plausible BI/analytics bands are set to NULL (even without salary data, the post itself e.g. skills are useful)
    -- WHERE Annual band:  $20,000 – $500,000
    -- WHERE Hourly band:  $10  – $250
    -- WHERE NOT min > max (data entry error)
    -- WHERE NOT both are set to NULL.
    salaries AS (
        SELECT *,

            CASE
                WHEN CAST(annual_salary_min AS DOUBLE) BETWEEN 20000 AND 500000
                 AND CAST(annual_salary_max AS DOUBLE) BETWEEN 20000 AND 500000
                 AND CAST(annual_salary_min AS DOUBLE) <= CAST(annual_salary_max AS DOUBLE)
                    THEN CAST(annual_salary_min AS DOUBLE)
                ELSE NULL
            END AS annual_salary_min_clean,

            CASE
                WHEN CAST(annual_salary_min AS DOUBLE) BETWEEN 20000 AND 500000
                 AND CAST(annual_salary_max AS DOUBLE) BETWEEN 20000 AND 500000
                 AND CAST(annual_salary_min AS DOUBLE) <= CAST(annual_salary_max AS DOUBLE)
                    THEN CAST(annual_salary_max AS DOUBLE)
                ELSE NULL
            END AS annual_salary_max_clean,

            CASE
                WHEN CAST(hourly_salary_min AS DOUBLE) BETWEEN 10 AND 250
                 AND CAST(hourly_salary_max AS DOUBLE) BETWEEN 10 AND 250
                 AND CAST(hourly_salary_min AS DOUBLE) <= CAST(hourly_salary_max AS DOUBLE)
                    THEN CAST(hourly_salary_min AS DOUBLE)
                ELSE NULL
            END AS hourly_salary_min_clean,

            CASE
                WHEN CAST(hourly_salary_min AS DOUBLE) BETWEEN 10 AND 250
                 AND CAST(hourly_salary_max AS DOUBLE) BETWEEN 10 AND 250
                 AND CAST(hourly_salary_min AS DOUBLE) <= CAST(hourly_salary_max AS DOUBLE)
                    THEN CAST(hourly_salary_max AS DOUBLE)
                ELSE NULL
            END AS hourly_salary_max_clean

        FROM labeled
    )

    -- ── FINAL SELECT ─────────────────────────────────────────────────
    -- Replace raw categorical and salary columns with cleaned versions.
    -- All other optional fields (city, soc_code, bls_*, skills, etc.)
    SELECT
        job_id,
        posted_at,
        title,
        normalized_title,
        company_name,
        location_text,
        city,
        city_code,
        state,
        state_code,
        remote_status_clean                          AS remote_status,
        employment_type_clean                        AS employment_type,
        salary_text,
        annual_salary_min_clean                      AS annual_salary_min,
        annual_salary_max_clean                      AS annual_salary_max,
        hourly_salary_min_clean                      AS hourly_salary_min,
        hourly_salary_max_clean                      AS hourly_salary_max,
        soc_code,
        soc_name,
        occupation_family,
        naics_code,
        naics_name,
        onet_code,
        onet_name,
        cip_code,
        cip_name,
        skills,
        program_matches,
        description,
        bls_employment,
        bls_annual_mean_wage,
        bls_annual_median_wage,
        bls_source_year,
        student_signals_work_type,
        student_signals_requirements_summary,
        student_signals_responsibilities_summary,
        student_signals_technical_tools_summary

    FROM salaries

""")
